"""OCR 후보만 추출한다. 미정 금액·총액은 계산해서 채우지 않는다."""
import math
import re
import unicodedata

MAX_AMOUNT = 2**53 - 1
MONEY = re.compile(r"(?:0|[1-9][0-9]*|[1-9][0-9]{0,2}(?:,[0-9]{3})+)(?:\s*원)?\Z")
TOTAL_LABELS = ("합계금액", "총결제금액", "총결제", "총합계", "결제금액", "받을금액", "승인금액", "청구금액", "합계", "총액")
TOTAL = re.compile(r"^(?:(?:신용카드|카드|현금)\s*)?(" + "|".join(r"\s*".join(label) for label in TOTAL_LABELS) + r")\s*[:：]?\s*(.*)$")
IGNORE = re.compile(r"사업자|등록번호|대표|전화|TEL|주소|가맹점|승인번호|카드번호|할부|부가세|과세|면세|공급가|받은돈|거스름돈|단말기|전표|바코드|상품코드", re.I)
PAYMENT = re.compile(r"^(?:신용카드|체크카드|카드|현금|간편결제|CASH|CREDIT|VISA|MASTER|VAT|TAX|SUBTOTAL|소계|중간합계|총수량|합계수량)", re.I)
FOREIGN = re.compile(r"\b(?:USD|EUR|JPY|CNY|RMB)\b|[$€¥]|달러|엔화|위안", re.I)
NOISE = re.compile(r"[0-9]{2,4}[-./년][ ]*[0-9]{1,2}[-./월][ ]*[0-9]{1,2}|[0-9]{2,3}-[0-9]{2,4}-[0-9]{4,5}|[0-9]{4}[-*]+[0-9]{4}")
ADJUSTMENT = re.compile(r"할인|환불|취소|반품|쿠폰|적립|포인트|[−-][ ]*[0-9]|[0-9]+\.[0-9]+|[0-9]+%")
HEADER = re.compile(r"품목명|상품명|메뉴명|품명|품목|상품|메뉴|단가|수량|금액")


def parse_amount(value):
    text = value.strip()
    if text.startswith("₩"):
        text = text[1:].strip()
    if not MONEY.fullmatch(text):
        return None
    digits = text.removesuffix("원").strip().replace(",", "")
    if len(digits) > 16:
        return None
    amount = int(digits)
    return amount if amount <= MAX_AMOUNT else None


def _box(block):
    value = block.get("box")
    if value is None:
        value = block.get("bbox")
    try:
        if not isinstance(value, (list, tuple)) or len(value) != 4:
            return None
        if isinstance(value[0], (list, tuple)):
            xs, ys = zip(*value)
            value = [min(xs), min(ys), max(xs), max(ys)]
        value = [float(v) for v in value]
        x1, y1, x2, y2 = value
        if not all(math.isfinite(v) for v in value) or x2 <= x1 or y2 <= y1:
            return None
        return value
    except (TypeError, ValueError):
        return None


def _lines(output):
    raw = output.get("text")
    raw = raw if isinstance(raw, str) else ""
    blocks = output.get("blocks")
    blocks = blocks if isinstance(blocks, list) else []
    blocks = [b for b in blocks if isinstance(b, dict) and isinstance(b.get("text"), str) and b["text"].strip()]
    positioned = [(b["text"].strip(), _box(b)) for b in blocks]
    if not positioned or any(box is None for _, box in positioned):
        return raw.splitlines() if raw.strip() else [text for text, _ in positioned]
    rows = []
    for text, box in sorted(positioned, key=lambda p: ((p[1][1] + p[1][3]) / 2, p[1][0])):
        center, height = (box[1] + box[3]) / 2, box[3] - box[1]
        if rows and abs(rows[-1]["y"] - center) <= 0.4 * min(height, rows[-1]["h"]):
            rows[-1]["parts"].append((box[0], text))
        else:
            rows.append({"y": center, "h": height, "parts": [(box[0], text)]})
    return [" ".join(text for _, text in sorted(row["parts"])) for row in rows]


def extract_receipt_fields(ocr_output):
    lines = [re.sub(r"\s+", " ", unicodedata.normalize("NFKC", line)).strip()
             for line in _lines(ocr_output)]
    items, totals, unparsed, warnings = [], [], [], []
    columns, pending_total = None, None
    for line in filter(None, lines):
        compact = line.replace(" ", "")
        if pending_total:
            amount = parse_amount(line)
            if amount is not None:
                totals.append({"amount": amount, "source": pending_total + " / " + line})
                pending_total = None
                continue
            unparsed.append(pending_total)
            pending_total = None
        if ADJUSTMENT.search(line) or FOREIGN.search(line):
            unparsed.append(line)
            warnings.append("할인·환불·소수·외화 등 자동 적용하지 않은 줄이 있습니다.")
            continue
        match = TOTAL.fullmatch(line)
        if match:
            tail = match.group(2)
            if not tail:
                pending_total = line
            else:
                amount = parse_amount(tail)
                if amount is None:
                    unparsed.append(line)
                else:
                    totals.append({"amount": amount, "source": line})
            continue
        if IGNORE.search(compact) or NOISE.search(line) or PAYMENT.search(compact):
            continue
        header = HEADER.findall(compact)
        if header and header[0] not in {"단가", "수량", "금액"} and "금액" in header and not re.search(r"[0-9]", line):
            columns = header[1:]
            continue
        tokens = line.split()
        if len(tokens) >= 2 and tokens[-1] == "원":
            tokens[-2:] = [tokens[-2] + "원"]
        numeric = []
        while tokens and parse_amount(tokens[-1]) is not None:
            numeric.insert(0, tokens.pop())
        name = " ".join(tokens).strip()
        if not name or not re.search(r"[가-힣A-Za-z]", name) or not numeric:
            unparsed.append(line)
            continue
        if any(re.fullmatch(r"[0-9OolI,./+−-]+(?:원|개)?", token) for token in tokens):
            unparsed.append(line)
            continue
        if len(numeric) == 1 and (columns is None or columns == ["금액"]):
            amount = parse_amount(numeric[0])
        elif columns and len(numeric) == len(columns) and columns.count("금액") == 1 and set(columns) <= {"단가", "수량", "금액"}:
            amount = parse_amount(numeric[columns.index("금액")])
        else:
            unparsed.append(line)
            warnings.append("단가·수량·항목 금액의 구분이 불명확한 줄은 직접 입력해 주세요.")
            continue
        items.append({"name": name, "amount": amount, "source": line})
    if pending_total:
        unparsed.append(pending_total)
    unique = {candidate["amount"] for candidate in totals}
    total = next(iter(unique)) if len(unique) == 1 else None
    if len(unique) > 1:
        warnings.append("서로 다른 총액 후보가 있습니다. 사진에서 최종 금액을 확인해 주세요.")
    if total is None:
        warnings.append("영수증 총액을 자동 확정하지 않았습니다. 품목 합계로 대신 채우지 않습니다.")
    return {"items": items, "total_amount": total, "total_candidates": totals,
            "unparsed_lines": unparsed, "warnings": list(dict.fromkeys(warnings))}

"""확인한 OCR 입력의 형식·원문·정산 초안을 한곳에서 만든다."""
from typing import Annotated, Literal
from pydantic import BaseModel, ConfigDict, Field, StrictBool, field_validator, model_validator

MAX_AMOUNT = 2**53 - 1
Amount = Annotated[int, Field(strict=True, ge=0, le=MAX_AMOUNT)]
Name = Annotated[str, Field(strict=True, min_length=1, max_length=100)]


class ReceiptItemInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    name: str | None = Field(default=None, max_length=200)
    amount: Amount | None = None

    @field_validator("name")
    @classmethod
    def trim_name(cls, value):
        return (value.strip() or None) if value is not None else None


class ReceiptDraftInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    expected_review_revision: int = Field(default=0, ge=0, strict=True)
    reviewed_by: Name
    image_checked: StrictBool
    items: list[ReceiptItemInput] = Field(max_length=300)
    total_amount: Amount | None = None
    receipt_total_amount: Amount | None = None
    participants: list[Name] = Field(default_factory=list, max_length=100)
    payer_name: Name | None = None
    split_method: Literal["equal", "explicit_amount"] | None = None
    note: str = Field(default="", max_length=2000)

    @field_validator("reviewed_by", "payer_name")
    @classmethod
    def validate_name(cls, value):
        if value is None:
            return None
        if not value.strip():
            raise ValueError("이름을 공백으로 입력할 수 없습니다.")
        return value.strip()

    @field_validator("participants")
    @classmethod
    def validate_people(cls, values):
        values = [v.strip() for v in values]
        if any(not v for v in values) or len(values) != len(set(values)):
            raise ValueError("분담 대상자의 빈 이름·중복 이름을 확인해 주세요.")
        return values

    @model_validator(mode="after")
    def validate_review(self):
        if not self.image_checked:
            raise ValueError("사진과 입력값을 대조한 뒤 확인해 주세요.")
        if sum(i.amount or 0 for i in self.items) > MAX_AMOUNT:
            raise ValueError("품목 합계가 정확히 다룰 수 있는 정수 범위를 넘습니다.")
        return self

    def content(self):
        return self.model_dump(exclude={"expected_review_revision"})


def manual_context(body):
    names = list(body.participants)
    # 결제자는 관련 인물에만 등록한다. 분담 대상으로 추가하지 않는다.
    if body.payer_name and body.payer_name not in names:
        names.append(body.payer_name)
    return {
        "participants": names,
        "expenses": [{"payer_name": body.payer_name, "total_amount": body.total_amount,
            "receipt_total_amount": body.receipt_total_amount,
            "items": [item.model_dump() for item in body.items]}],
        "split_rules": [{"expense_index": 0, "item_index": index,
            "method": body.split_method, "participant_names": body.participants or None}
            for index in range(len(body.items))],
    }


def reviewed_evidence(body):
    def money(value):
        return "미정" if value is None else f"{value:,}원"
    lines = ["[사용자가 대조한 영수증 입력]"]
    lines.extend(f"품목 {index + 1}: {item.name or '품목명 미정'} / 항목 전체 금액: {money(item.amount)}"
                 for index, item in enumerate(body.items))
    lines.extend([f"영수증에 표시된 총액: {money(body.receipt_total_amount)}",
                  f"사용자가 입력한 실제 결제 총액: {money(body.total_amount)}"])
    if body.note.strip():
        lines.append("추가 확인 메모: " + body.note.strip())
    evidence = "\n".join(lines)
    if len(evidence) > 24000:
        raise ValueError("영수증 근거가 24,000자를 넘습니다. 품목명과 메모를 줄여 주세요.")
    return evidence


def condition_text(body):
    lines = []
    if body.participants:
        lines.append("비용 분담 대상: " + ", ".join(body.participants))
    if body.payer_name:
        lines.append("먼저 결제한 사람: " + body.payer_name)
    if body.split_method == "equal":
        lines.append("입력한 분담 대상자가 각 품목을 균등 분담하기로 선택함.")
    elif body.split_method == "explicit_amount":
        lines.append("개인별 금액 지정 방식. 구체적인 개인별 금액은 추가 확인 필요.")
    return "\n".join(lines) or "영수증에서 직접 입력한 정산. 결제자와 분담 조건은 추가 확인 필요."

"""로컬 단일 프로세스용 영수증 업로드 → 사람 검수 → 기존 LLM/정산 연결."""
from contextlib import closing
from datetime import datetime, timezone
from hashlib import sha256
from pathlib import Path
from threading import Lock
from types import SimpleNamespace
from typing import Literal
from uuid import uuid4
import json
import sqlite3
import time

from fastapi import APIRouter, File, HTTPException, UploadFile
from fastapi.responses import FileResponse
from pydantic import BaseModel, ConfigDict, Field, StrictBool

from .parser import load_contract, validate_context_data
from .receipt_ocr import MAX_BYTES, OCRError, PaddleReceiptOCR, prepare_image
from .receipt_fields import extract_receipt_fields
from .receipt_draft import ReceiptDraftInput, manual_context, reviewed_evidence, condition_text


class ReceiptInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    reviewed_text: str = Field(min_length=1, max_length=24000)
    reviewed_by: str = Field(min_length=1, max_length=100)
    image_checked: StrictBool
    text: str = Field(min_length=1, max_length=12000)
    mode: Literal["llm", "manual"] = "llm"


class ReceiptAIInput(BaseModel):
    model_config = ConfigDict(extra="forbid")
    expected_review_revision: int = Field(ge=1, strict=True)
    text: str = Field(min_length=1, max_length=12000)


def receipt_router(store, parse, root, ocr=None):
    router = APIRouter()
    engine = ocr if ocr is not None else PaddleReceiptOCR()
    directory = Path(store.path).parent / "receipts"
    directory.mkdir(parents=True, exist_ok=True)
    upload_lock, parse_lock = Lock(), Lock()
    last_call = [float("-inf")]
    with closing(sqlite3.connect(store.path)) as db:
        db.execute("CREATE TABLE IF NOT EXISTS receipts (id TEXT PRIMARY KEY, document TEXT NOT NULL)")
        db.commit()

    def save(doc, new=False):
        with closing(sqlite3.connect(store.path)) as db:
            value = json.dumps(doc, ensure_ascii=False, allow_nan=False)
            if new:
                db.execute("INSERT INTO receipts VALUES (?, ?)", (doc["receipt_id"], value))
            else:
                db.execute("UPDATE receipts SET document = ? WHERE id = ?", (value, doc["receipt_id"]))
            db.commit()

    def get(rid):
        with closing(sqlite3.connect(store.path)) as db:
            row = db.execute("SELECT document FROM receipts WHERE id = ?", (rid,)).fetchone()
        if row is None:
            raise HTTPException(404, "영수증을 찾지 못했습니다.")
        doc = json.loads(row[0])
        doc.setdefault("review_revision", 0)
        if "fields" not in doc:
            doc["fields"] = extract_receipt_fields(doc.get("ocr") or {})
        return doc

    def request_key(value):
        return sha256(json.dumps(value, sort_keys=True, ensure_ascii=False).encode()).hexdigest()

    def save_in(db, doc):
        db.execute("UPDATE receipts SET document = ? WHERE id = ?",
                   (json.dumps(doc, ensure_ascii=False, allow_nan=False), doc["receipt_id"]))

    def read_in(db, rid):
        row = db.execute("SELECT document FROM receipts WHERE id = ?", (rid,)).fetchone()
        if row is None:
            raise HTTPException(404, "영수증을 찾지 못했습니다.")
        return json.loads(row[0])

    def review_value(body):
        try:
            evidence = reviewed_evidence(body)
        except ValueError as exc:
            raise HTTPException(422, str(exc)) from exc
        return {**body.content(), "reviewed_text": evidence,
                "reviewed_at": datetime.now(timezone.utc).isoformat()}

    def check_version(doc, expected):
        if doc.get("review_revision", 0) != expected:
            raise HTTPException(409, "다른 화면에서 영수증이 수정됐습니다. 새로 불러와 대조해 주세요.")

    def manual_parsed(body, evidence):
        contract = load_contract()
        return SimpleNamespace(context=validate_context_data(manual_context(body)),
            source_text=condition_text(body), supplied_evidence=evidence, model="manual",
            prompt_version=contract[2], schema_sha256=contract[3], attempts=0)

    def commit_session(rid, parsed, key, expected, review=None):
        with closing(sqlite3.connect(store.path, timeout=10)) as db:
            with db:
                db.execute("BEGIN IMMEDIATE")
                doc = read_in(db, rid)
                if doc["session_id"]:
                    if doc["request_key"] != key:
                        raise HTTPException(409, "이미 연결한 영수증입니다. 연결한 정산에서 수정해 주세요.")
                    sid = doc["session_id"]
                else:
                    check_version(doc, expected)
                    if review is not None:
                        doc["review"] = review
                        doc["review_revision"] = expected + 1
                    sid = store.start(parsed, connection=db)
                    doc.update(session_id=sid, request_key=key)
                    save_in(db, doc)
        return store.view(sid)

    @router.post("/api/receipts/{rid}/draft")
    def manual_draft(rid: str, body: ReceiptDraftInput):
        review = review_value(body)
        parsed = manual_parsed(body, review["reviewed_text"])
        return commit_session(rid, parsed, request_key({"mode": "manual", **body.content()}),
                              body.expected_review_revision, review)

    @router.post("/api/receipts/{rid}/review")
    def prepare_ai(rid: str, body: ReceiptDraftInput):
        review = review_value(body)
        key = request_key(body.content())
        with closing(sqlite3.connect(store.path, timeout=10)) as db:
            with db:
                db.execute("BEGIN IMMEDIATE")
                doc = read_in(db, rid)
                if doc["session_id"]:
                    raise HTTPException(409, "이미 연결한 영수증입니다. 연결한 정산에서 수정해 주세요.")
                if doc.get("review_key") != key:
                    check_version(doc, body.expected_review_revision)
                    doc.update(review=review, review_key=key,
                               review_revision=body.expected_review_revision + 1)
                    save_in(db, doc)
        return {"receipt_id": rid, "review_revision": doc["review_revision"],
                "evidence": doc["review"]["reviewed_text"], "hint": condition_text(body)}

    @router.get("/api/receipts/{rid}/handoff")
    def ai_handoff(rid: str):
        doc = get(rid)
        if not doc.get("review_key"):
            raise HTTPException(409, "영수증 입력 화면에서 먼저 검수해 주세요.")
        review = doc["review"]
        data = {k: review[k] for k in ReceiptDraftInput.model_fields if k in review}
        return {"receipt_id": rid, "review_revision": doc["review_revision"],
                "evidence": review["reviewed_text"], "hint": condition_text(ReceiptDraftInput(**data))}

    @router.post("/api/receipts/{rid}/ai")
    def apply_ai(rid: str, body: ReceiptAIInput):
        if not body.text.strip():
            raise HTTPException(422, "정산 조건을 입력해 주세요.")
        if not parse_lock.acquire(blocking=False):
            raise HTTPException(429, "다른 AI 추출을 처리 중입니다.")
        try:
            doc = get(rid)
            check_version(doc, body.expected_review_revision)
            if not doc.get("review_key"):
                raise HTTPException(409, "영수증 검수가 필요합니다.")
            key = request_key({"mode": "llm", "review_key": doc["review_key"], "text": body.text})
            if doc["session_id"]:
                if doc["request_key"] != key:
                    raise HTTPException(409, "이미 연결한 영수증입니다.")
                return store.view(doc["session_id"])
            if time.monotonic() - last_call[0] < 10:
                raise HTTPException(429, "AI 재호출은 10초 후 가능합니다.")
            last_call[0] = time.monotonic()
            parsed = parse(body.text, supplied_evidence=doc["review"]["reviewed_text"])
            return commit_session(rid, parsed, key, body.expected_review_revision)
        finally:
            parse_lock.release()

    @router.get("/receipts/new")
    def page():
        return FileResponse(Path(root) / "frontend" / "receipt.html")

    @router.post("/api/receipts", status_code=201)
    def upload(file: UploadFile = File(...)):
        if not upload_lock.acquire(blocking=False):
            raise HTTPException(429, "다른 영수증을 읽는 중입니다. 잠시 후 다시 시도해 주세요.")
        try:
            try:
                raw = file.file.read(MAX_BYTES + 1)
                image = prepare_image(raw)
            except ValueError as exc:
                raise HTTPException(422, str(exc)) from exc
            rid = uuid4().hex
            folder = directory / rid
            folder.mkdir()
            (folder / "original.bin").write_bytes(raw)
            image.save(folder / "image.png")
            warning = None
            try:
                output = engine.read(folder / "image.png")
            except OCRError as exc:
                warning = str(exc)
                output = {"text": "", "blocks": [], "low_score_count": 0}
            if not output["text"] and not warning:
                warning = "글자를 찾지 못했습니다. 선명한 사진을 올리거나 직접 입력해 주세요."
            doc = {"receipt_id": rid, "created_at": datetime.now(timezone.utc).isoformat(),
                   "image_sha256": sha256(raw).hexdigest(), "ocr": output,
                   "fields": extract_receipt_fields(output), "review_revision": 0,
                   "warning": warning, "review": None, "request_key": None, "session_id": None}
            save(doc, new=True)
            return doc
        finally:
            file.file.close()
            upload_lock.release()

    @router.get("/api/receipts/{rid}")
    def read(rid: str):
        return get(rid)

    @router.get("/api/receipts/{rid}/image")
    def image(rid: str):
        doc = get(rid)
        return FileResponse(directory / doc["receipt_id"] / "image.png", media_type="image/png")

    @router.post("/api/receipts/{rid}/session")
    def start_session(rid: str, body: ReceiptInput):
        # 기존 클라이언트용 경로. 새 화면은 /draft 또는 검수 후 /ai를 사용한다.
        if not body.image_checked or not all(v.strip() for v in
                (body.reviewed_text, body.reviewed_by, body.text)):
            raise HTTPException(422, "영수증 대조 확인과 담당자·텍스트·정산 조건을 입력해 주세요.")
        if not parse_lock.acquire(blocking=False):
            raise HTTPException(429, "다른 추출을 처리 중입니다. 잠시 후 다시 시도해 주세요.")
        try:
            doc = get(rid)
            if doc.get("review_key"):
                raise HTTPException(409, "검수 화면의 정산하기 또는 AI 적용 버튼을 이용해 주세요.")
            key = sha256(json.dumps(body.model_dump(), sort_keys=True, ensure_ascii=False).encode()).hexdigest()
            if doc["session_id"]:
                if doc["request_key"] != key:
                    raise HTTPException(409, "이미 연결한 영수증입니다. 정산 화면에서 조건을 수정하거나 새 영수증으로 시작해 주세요.")
                return store.view(doc["session_id"])
            if body.mode == "llm" and time.monotonic() - last_call[0] < 10:
                raise HTTPException(429, "AI 재호출 간격은 10초입니다. 잠시 후 다시 시도하거나 직접 입력해 주세요.")
            # 실패하더라도 원본 OCR과 사람이 수정한 근거는 각각 보관한다.
            expected = doc.get("review_revision", 0)
            with closing(sqlite3.connect(store.path, timeout=10)) as db:
                with db:
                    db.execute("BEGIN IMMEDIATE")
                    current = read_in(db, rid)
                    check_version(current, expected)
                    if current["session_id"]:
                        raise HTTPException(409, "이미 연결한 영수증입니다.")
                    current["review"] = {**body.model_dump(), "reviewed_at": datetime.now(timezone.utc).isoformat()}
                    current["review_revision"] = expected + 1
                    save_in(db, current)
            if body.mode == "llm":
                last_call[0] = time.monotonic()
                parsed = parse(body.text, supplied_evidence=body.reviewed_text)
            else:
                contract = load_contract()
                parsed = SimpleNamespace(context=validate_context_data({"participants": [], "expenses": []}),
                    source_text=body.text, supplied_evidence=body.reviewed_text, model="manual",
                    prompt_version=contract[2], schema_sha256=contract[3], attempts=0)
            # start()는 검토할 초안만 만든다. 확인/계산은 기존 workflow의 별도 단계다.
            return commit_session(rid, parsed, key, expected + 1)
        finally:
            parse_lock.release()

    @router.get("/api/sessions/{sid}/receipt")
    def session_receipt(sid: str):
        with closing(sqlite3.connect(store.path)) as db:
            rows = db.execute("SELECT document FROM receipts").fetchall()
        for (value,) in rows:
            doc = json.loads(value)
            if doc["session_id"] == sid:
                return {"receipt_id": doc["receipt_id"], "review": doc["review"]}
        return None

    return router

"""웹 요청 사이에 정산 초안·확인·계산 버전을 보존하는 저장 모듈.

SQLite에 원문, 모든 입력 버전, 확인 기록, 계산 미리보기를 보존한다.
은행 송금·메시지 발송·입금 완료 처리는 하지 않는다.
"""
from contextlib import contextmanager
from dataclasses import asdict
from datetime import datetime, timezone
from hashlib import sha256
from pathlib import Path
import json
import sqlite3
from uuid import uuid4

from .calculator import CalculationError, calculate_settlement
from .parser import ParserError, load_contract, parse_and_review, review_context, validate_context_data

PROJECT_ROOT = Path(__file__).resolve().parents[2]


class WorkflowError(ValueError):
    pass


def encode(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, allow_nan=False)


def digest(value):
    return sha256(encode(value).encode("utf-8")).hexdigest()


def now():
    return datetime.now(timezone.utc).isoformat()


def review(data):
    schema = load_contract()[1]
    clean = validate_context_data(data, schema=schema).model_dump(exclude_unset=True)
    issues = [issue.model_dump() for issue in review_context(clean, schema=schema)]
    return clean, issues


class WorkflowStore:
    def __init__(self, path):
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with self._connection() as db:
            db.execute("CREATE TABLE IF NOT EXISTS sessions (id TEXT PRIMARY KEY, document TEXT NOT NULL)")
            db.commit()

    @contextmanager
    def _connection(self):
        db = sqlite3.connect(self.path, timeout=10)
        try:
            yield db
        finally:
            db.close()

    @contextmanager
    def _transaction(self, session_id):
        with self._connection() as db:
            db.execute("BEGIN IMMEDIATE")
            try:
                row = db.execute("SELECT document FROM sessions WHERE id = ?", (session_id,)).fetchone()
                if row is None:
                    raise WorkflowError("존재하지 않는 세션입니다.")
                document = json.loads(row[0])
                yield document
                db.execute("UPDATE sessions SET document = ? WHERE id = ?", (encode(document), session_id))
                db.commit()
            except Exception:
                db.rollback()
                raise

    def _revision(self, data, number):
        clean, issues = review(data)
        return {"input_revision": number, "context": clean, "issues": issues,
                "context_sha256": digest(clean), "schema_sha256": load_contract()[3],
                "created_at": now()}

    def start(self, parsed, *, connection=None):
        session_id = uuid4().hex
        revision = self._revision(parsed.context.model_dump(exclude_unset=True), 1)
        if parsed.schema_sha256 != revision["schema_sha256"]:
            raise WorkflowError("추출에 사용한 스키마와 현재 스키마가 다릅니다.")
        document = {
            "session_id": session_id,
            "source_text": parsed.source_text,
            "supplied_evidence": parsed.supplied_evidence,
            "parse_metadata": {key: getattr(parsed, key) for key in
                               ("model", "prompt_version", "schema_sha256", "attempts")},
            "revisions": [revision], "confirmations": [], "calculations": [],
        }
        if connection is not None:
            # 영수증과 정산 초안을 같은 트랜잭션에서 저장할 때 사용한다.
            connection.execute("INSERT INTO sessions VALUES (?, ?)", (session_id, encode(document)))
        else:
            with self._connection() as db:
                db.execute("INSERT INTO sessions VALUES (?, ?)", (session_id, encode(document)))
                db.commit()
        return session_id

    def view(self, session_id):
        with self._connection() as db:
            row = db.execute("SELECT document FROM sessions WHERE id = ?", (session_id,)).fetchone()
        if row is None:
            raise WorkflowError("존재하지 않는 세션입니다.")
        return json.loads(row[0])

    def revise(self, session_id, expected_revision, edited_context):
        with self._transaction(session_id) as doc:
            current = doc["revisions"][-1]
            if current["input_revision"] != expected_revision:
                raise WorkflowError("입력 버전이 변경됐습니다. 최신 내용을 다시 확인해 주세요.")
            revision = self._revision(edited_context, expected_revision + 1)
            doc["revisions"].append(revision)
            # 과거 확인·계산은 삭제하지 않는다. 버전 불일치로 재사용을 막는다.
            return revision

    def confirm(self, session_id, expected_revision, *, reviewed_by, review_notes, checks):
        if not isinstance(reviewed_by, str) or not reviewed_by.strip():
            raise WorkflowError("확인 담당자를 입력해 주세요.")
        if not isinstance(review_notes, str) or not review_notes.strip():
            raise WorkflowError("원문·조건 확인 근거를 기록해 주세요.")
        if not isinstance(checks, dict):
            raise WorkflowError("확인 항목이 필요합니다.")
        with self._transaction(session_id) as doc:
            revision = doc["revisions"][-1]
            if revision["input_revision"] != expected_revision:
                raise WorkflowError("이전 버전은 확인할 수 없습니다.")
            self._check_revision(revision)
            required = {"source_reviewed", "people_resolved", "conditions_reviewed"}
            if revision["context"].get("contributions"):
                required.add("funding_verified")
            if any(checks.get(key) is not True for key in required):
                raise WorkflowError("원문·실제 인물·분담 조건·필요한 찬조 자금 확인을 완료해 주세요.")
            _, issues = review(revision["context"])
            if issues:
                raise WorkflowError(encode(issues))
            for record in doc["confirmations"]:
                if record["input_revision"] == expected_revision:
                    return record
            record = {
                "confirmation_id": uuid4().hex, "input_revision": expected_revision,
                "context_sha256": revision["context_sha256"],
                "reviewed_by": reviewed_by.strip(), "review_notes": review_notes.strip(),
                "checks": {key: True for key in sorted(required)}, "confirmed_at": now(),
            }
            doc["confirmations"].append(record)
            return record

    def _check_revision(self, revision):
        if digest(revision["context"]) != revision["context_sha256"]:
            raise WorkflowError("저장된 입력의 해시가 일치하지 않습니다.")
        if revision["schema_sha256"] != load_contract()[3]:
            raise WorkflowError("스키마가 바뀌었습니다. 새 입력 버전으로 검토해 주세요.")

    def calculate(self, session_id, confirmation_id):
        with self._transaction(session_id) as doc:
            record = next((c for c in doc["confirmations"]
                           if c["confirmation_id"] == confirmation_id), None)
            if record is None:
                raise WorkflowError("이 세션의 유효한 확인 기록이 아닙니다.")
            revision = doc["revisions"][-1]
            self._check_revision(revision)
            if (record["input_revision"] != revision["input_revision"]
                    or record["context_sha256"] != revision["context_sha256"]):
                raise WorkflowError("확인 이후 입력이 변경됐습니다. 다시 확인해 주세요.")
            for saved in doc["calculations"]:
                if saved["confirmation_id"] == confirmation_id:
                    return saved
            result = calculate_settlement(revision["context"])
            saved = {
                "calculation_id": uuid4().hex, "confirmation_id": confirmation_id,
                "input_revision": revision["input_revision"], "status": "preview",
                "created_at": now(), "result": asdict(result),
            }
            doc["calculations"].append(saved)
            return saved
